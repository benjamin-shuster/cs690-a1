def summarize_ranges(nums):
    result = []
    if not nums:
        return result

    start = prev = nums[0]
    for num in nums[1:]:
        if num == prev + 1:
            prev = num
        else:
            result.append(str(start) if start == prev else f"{start}-{prev}")
            start = prev = num

    result.append(str(start) if start == prev else f"{start}-{prev}")
    return result
